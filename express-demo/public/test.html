The export button in your React app

// Express: load once, render per request
const tpl = Template.loadSync(fs.readFileSync("report.xlsx"));
app.get("/api/report.xlsx", async (req, res) => {
  const rows = await db.query("SELECT ...");
  const xlsx = await tpl.render({ Table1: rows });
  res.attachment("report.xlsx").send(Buffer.from(xlsx));
});
